# Purrgotchi v3.4.6 — C2 + C3 + C4 Commands Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship Purrgotchi v3.4.6 — adds `purrgotchi play` (coin-costing bond bump, 6h cooldown), `purrgotchi treat` (coin-costing hunger reset + minor bond, 4h cooldown), and `purrgotchi stats` (curated human-readable state view). First coin-spending sinks in the game.

**Architecture:** All interaction logic stays in `bond.rs` — two new `try_X` functions + result enums mirror the existing `try_pet` shape. Two additive `serde(default)` state fields (`last_played_at`, `last_treated_at`) — no schema version bump. `compute_mood` grows to accept three pet-like deltas, OR'd against their own window constants, converging on the existing `happy` sprite. Bond decay semantics corrected to anchor on the most recent interaction of any kind via a new `last_interaction_at(&State)` helper.

**Tech Stack:** Rust 2024, `serde`, `serde_json`.

**Spec:** `docs/superpowers/specs/2026-04-24-c2-c3-c4-commands-design.md`

```rust
#[derive(Debug, PartialEq)]
pub enum PlayResult {
    Applied { old: u8, new: u8, capped: bool, cost: u64 },
    Cooldown { wait_secs: u64 },
    InsufficientCoins { need: u64, have: u64 },
}

pub fn try_play(state: &mut State, now: u64) -> PlayResult {
    // Gate 1: cooldown
    if state.last_played_at != 0 {
        let elapsed = now.saturating_sub(state.last_played_at);
        if elapsed < PLAY_COOLDOWN_SEC {
            return PlayResult::Cooldown {
                wait_secs: PLAY_COOLDOWN_SEC - elapsed,
            };
        }
    }
    // Gate 2: coins
    if state.coins < PLAY_COST {
        return PlayResult::InsufficientCoins {
            need: PLAY_COST,
            have: state.coins,
        };
    }
    // Apply
    let anchor = last_interaction_at(state);
    let effective = effective_bond(state.bond, anchor, now);
    let capped = effective == 100;
    let new = effective.saturating_add(PLAY_BOND_BUMP).min(100);
    state.bond = new;
    state.last_played_at = now;
    state.coins -= PLAY_COST;
    PlayResult::Applied { old: effective, new, capped, cost: PLAY_COST }
}
```

```rust
#[derive(Debug, PartialEq)]
pub enum TreatResult {
    Applied {
        old: u8,
        new: u8,
        capped: bool,
        cost: u64,
        was_hungry: bool,
    },
    Cooldown { wait_secs: u64 },
    InsufficientCoins { need: u64, have: u64 },
}

pub fn try_treat(state: &mut State, now: u64) -> TreatResult {
    // Gate 1: cooldown
    if state.last_treated_at != 0 {
        let elapsed = now.saturating_sub(state.last_treated_at);
        if elapsed < TREAT_COOLDOWN_SEC {
            return TreatResult::Cooldown {
                wait_secs: TREAT_COOLDOWN_SEC - elapsed,
            };
        }
    }
    // Gate 2: coins
    if state.coins < TREAT_COST {
        return TreatResult::InsufficientCoins {
            need: TREAT_COST,
            have: state.coins,
        };
    }
    // Pre-check hunger (must be evaluated BEFORE resetting last_fed_at)
    let was_hungry = now.saturating_sub(state.last_fed_at) >= TREAT_HUNGER_THRESHOLD_SEC;

    let anchor = last_interaction_at(state);
    let effective = effective_bond(state.bond, anchor, now);
    let capped = effective == 100;
    let new = if was_hungry {
        effective.saturating_add(TREAT_BOND_BUMP).min(100)
    } else {
        effective
    };
    state.bond = new;
    state.last_treated_at = now;
    state.last_fed_at = now; // hunger reset regardless of was_hungry
    state.coins -= TREAT_COST;
    TreatResult::Applied { old: effective, new, capped, cost: TREAT_COST, was_hungry }
}
```

```rust
fn format_stats_card(s: &crate::state::State, now: u64) -> String {
    use std::fmt::Write;
    use crate::identity::{STAT_NAMES, rarity_stars};
    use crate::bond::{effective_bond, last_interaction_at};

    let mut out = String::new();
    let stars = rarity_stars(&s.rarity, s.shiny);

    // --- Identity ---
    writeln!(out, "┌─ {} {}", s.name, "─".repeat(46)).ok();
    writeln!(out, "  Identity").ok();
    writeln!(out, "    Name:       {}", s.name).ok();
    writeln!(out, "    Breed:      {} ({} {})", s.breed, s.rarity, stars).ok();
    writeln!(out, "    Born:       {}", format_born_relative(s.birth_time, now)).ok();
    writeln!(
        out,
        "    Life stage: {} (level {})",
        crate::state::life_stage(s.level),
        s.level,
    ).ok();
    writeln!(out).ok();

    // --- Stats ---
    writeln!(out, "  Stats").ok();
    let (peak_idx, dump_idx) = peak_dump_indices(&s.stats);
    for (i, name) in STAT_NAMES.iter().enumerate() {
        let val = s.stats[i];
        let bar = nine_cell_bar(val);
        let annotation = if i == peak_idx {
            "  \u{2190} peak"
        } else if i == dump_idx {
            "  \u{2190} dump"
        } else {
            ""
        };
        writeln!(out, "    {:<10} {} {}{}", name, bar, val, annotation).ok();
    }
    writeln!(out).ok();

    // --- Relationship ---
    writeln!(out, "  Relationship").ok();
    let eff = effective_bond(s.bond, last_interaction_at(s), now);
    writeln!(out, "    Bond:       {} {}", nine_cell_bar(eff), eff).ok();
    writeln!(out, "    Last pet:   {}", timestamp_elapsed(s.last_pet_at, now)).ok();
    writeln!(out, "    Last play:  {}", timestamp_elapsed(s.last_played_at, now)).ok();
    writeln!(out, "    Last treat: {}", timestamp_elapsed(s.last_treated_at, now)).ok();
    writeln!(out).ok();

    // --- Progress ---
    writeln!(out, "  Progress").ok();
    writeln!(out, "    Tool calls: {}", s.total_tool_calls).ok();
    writeln!(out, "    Coins:      {}", s.coins).ok();
    writeln!(out, "└{}", "─".repeat(48)).ok();

    out
}

fn nine_cell_bar(val: u8) -> String {
    let val = val.min(100) as usize;
    let filled = (val * 9 + 50) / 100;
    let filled = filled.min(9);
    let mut s = String::with_capacity(27);
    for _ in 0..filled { s.push('\u{2588}'); }
    for _ in 0..(9 - filled) { s.push('\u{2591}'); }
    s
}

fn peak_dump_indices(stats: &[u8; 5]) -> (usize, usize) {
    let mut peak = 0usize;
    let mut dump = 0usize;
    for i in 1..5 {
        if stats[i] > stats[peak] { peak = i; }
        if stats[i] < stats[dump] { dump = i; }
    }
    (peak, dump)
}

fn timestamp_elapsed(ts: u64, now: u64) -> String {
    if ts == 0 {
        return "never".to_string();
    }
    let delta = now.saturating_sub(ts) as i64;
    crate::render::human_elapsed(delta)
}

fn format_born_relative(birth_time: u64, now: u64) -> String {
    let date = format_born_date(birth_time);
    let elapsed_days = now.saturating_sub(birth_time) / 86400;
    match elapsed_days {
        0 => format!("{date} (today)"),
        1 => format!("{date} (yesterday)"),
        n => format!("{date} ({n} days ago)"),
    }
}

fn format_born_date(secs: u64) -> String {
    // Same algorithm as card::format_born, inlined to avoid making card::format_born pub.
    let mut days = (secs / 86400) as u32;
    let mut year = 1970u32;
    loop {
        let in_year = if is_leap(year) { 366 } else { 365 };
        if days < in_year { break; }
        days -= in_year;
        year += 1;
    }
    let month_days = [31u32, if is_leap(year) { 29 } else { 28 }, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31];
    let mut month = 0usize;
    for &md in &month_days {
        if days < md { break; }
        days -= md;
        month += 1;
    }
    format!("{year}-{:02}-{:02}", month + 1, days + 1)
}

fn is_leap(y: u32) -> bool {
    (y % 4 == 0 && y % 100 != 0) || y % 400 == 0
}
```
